GetInts <- function(region, dat) {
  res <- lapply(dat[[region]], function(lst) {
    df <- c(PO=lst$alpha, PA=c(lst$coefficients["(Intercept)"]))
    df
  })
  res <- t(list2DF(res))
  colnames(res) <- c("PO", "PA")
  res
}
